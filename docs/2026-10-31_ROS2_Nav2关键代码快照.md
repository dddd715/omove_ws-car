```python
import os
from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, IncludeLaunchDescription, OpaqueFunction
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node
from launch.substitutions import LaunchConfiguration
import xacro

def launch_setup(context):
    pkg_share = get_package_share_directory('omove_description')
    xacro_file = LaunchConfiguration('robot_file').perform(context)
    robot_description = xacro.process_file(xacro_file, mappings={
        'controllers_file': LaunchConfiguration('controllers_file').perform(context),
        'wheel_control_mode': LaunchConfiguration('wheel_control_mode').perform(context),
    }).toxml()

    # 发布TF和URDF
    rsp = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        output='screen',
        parameters=[{'robot_description': robot_description, 'use_sim_time': True}]
    )

    # 启动Gazebo
    gazebo = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(get_package_share_directory('gazebo_ros'),
                         'launch', 'gazebo.launch.py')
        ),
        launch_arguments={'world': LaunchConfiguration('world')}.items(),
    )

    # 把小车插入Gazebo
    spawn = Node(
        package='gazebo_ros',
        executable='spawn_entity.py',
        arguments=['-topic', 'robot_description', '-entity', 'omove',
                   '-x', LaunchConfiguration('x'), '-y', LaunchConfiguration('y'),
                   '-z', LaunchConfiguration('z')],
        output='screen'
    )
    # Spawners wait for controller_manager, which is created by the Gazebo plugin.
    controllers = [
        Node(
            package='controller_manager',
            executable='spawner',
            arguments=[controller, '--controller-manager', '/controller_manager',
                       '--controller-manager-timeout', '120'],
            output='screen',
        )
        for controller in [
            'joint_state_broadcaster',
            'front_left_velocity_controller',
            'front_right_velocity_controller',
            'rear_left_velocity_controller',
            'rear_right_velocity_controller',
        ]
    ]
    kinematics = Node(
        package='omove_description',
        executable='mecanum_kinematics.py',
        parameters=[{'use_sim_time': True}],
        output='screen',
    )

    odometry = Node(
        package='omove_description',
        executable='mecanum_odometry.py',
        parameters=[{'use_sim_time': True}],
        output='screen',
    )

    return [rsp, gazebo, spawn, kinematics, odometry] + controllers


def generate_launch_description():
    share = get_package_share_directory('omove_description')
    return LaunchDescription([
        DeclareLaunchArgument('world', default_value=os.path.join(get_package_share_directory('gazebo_ros'), 'worlds', 'empty.world')),
        DeclareLaunchArgument('x', default_value='0.0'),
        DeclareLaunchArgument('y', default_value='0.0'),
        DeclareLaunchArgument('z', default_value='0.0'),
        DeclareLaunchArgument('controllers_file', default_value=os.path.join(share, 'config', 'mecanum_controllers.yaml')),
        DeclareLaunchArgument('robot_file', default_value=os.path.join(share, 'urdf', 'omove.urdf.xacro')),
        DeclareLaunchArgument('wheel_control_mode', default_value='velocity', choices=['velocity', 'velocity_pid']),
        OpaqueFunction(function=launch_setup)])
```

```python
"""Nav2 in world01; require an explicit map rather than loading the room map."""
import os
import socket
from urllib.parse import urlparse
from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, IncludeLaunchDescription, OpaqueFunction
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch.substitutions import LaunchConfiguration


def require_free_gazebo_master(context):
    """Fail before starting any robot/Nav2 nodes if an old simulation survives."""
    uri = urlparse(os.environ.get('GAZEBO_MASTER_URI', 'http://localhost:11345'))
    try:
        connection = socket.create_connection((uri.hostname or 'localhost', uri.port or 11345), timeout=1.)
    except ConnectionRefusedError:
        return []
    except OSError as error:
        raise RuntimeError(f'Cannot verify Gazebo master is free: {error}') from error
    connection.close()
    raise RuntimeError(
        'Gazebo master is already running. Stop the old OMOVE launch and its '
        'remaining gzserver/robot nodes before starting world01_nav; otherwise '
        'duplicate odometry and TF publishers can corrupt navigation.')


def generate_launch_description():
    share = get_package_share_directory('omove_description')
    return LaunchDescription([
        DeclareLaunchArgument('map', description='Absolute path to a saved world01 map YAML'),
        DeclareLaunchArgument('gui', default_value='true'),
        DeclareLaunchArgument('x', default_value='0.0'),
        DeclareLaunchArgument('y', default_value='-1.0'),
        DeclareLaunchArgument('params_file', default_value=os.path.join(share, 'config', 'nav2_params.yaml')),
        OpaqueFunction(function=require_free_gazebo_master),
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(os.path.join(share, 'launch', 'world01_sim.launch.py')),
            launch_arguments={'slam': 'False', 'gui': LaunchConfiguration('gui'),
                              'x': LaunchConfiguration('x'), 'y': LaunchConfiguration('y')}.items()),
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource(os.path.join(share, 'launch', 'nav_bringup.launch.py')),
            launch_arguments={'map': LaunchConfiguration('map'),
                              'params_file': LaunchConfiguration('params_file'),
                              'use_sim_time': 'true'}.items()),
    ])
```

```python
import os
from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch.substitutions import LaunchConfiguration

def generate_launch_description():
    pkg_share = get_package_share_directory('omove_description')
    params_file = LaunchConfiguration('params_file')
    map_file = LaunchConfiguration('map')

    nav2 = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(get_package_share_directory('nav2_bringup'),
                         'launch', 'bringup_launch.py')
        ),
        launch_arguments={
            # This entry always localizes on a saved map. Explicit Python-style
            # False also prevents an inherited 'slam=false' from breaking
            # Humble Nav2's PythonExpression(['not ', slam]).
            'slam': 'False',
            'use_sim_time': LaunchConfiguration('use_sim_time'),
            'use_composition': 'False',
            'autostart': 'true',
            'map': map_file,
            'params_file': params_file,
        }.items()
    )

    return LaunchDescription([
        DeclareLaunchArgument('map', default_value=os.path.join(pkg_share, 'maps', 'room.yaml')),
        DeclareLaunchArgument('params_file', default_value=os.path.join(pkg_share, 'config', 'nav2_params.yaml')),
        DeclareLaunchArgument('use_sim_time', default_value='true'),
        nav2,
    ])
```

```python
#!/usr/bin/env python3
import math

import rclpy
from rclpy.node import Node
from rclpy.clock import Clock, ClockType
from geometry_msgs.msg import Twist
from std_msgs.msg import Float64MultiArray

class MecanumKinematics(Node):
    def __init__(self):
        super().__init__('mecanum_kinematics')
        self.r = self.declare_parameter('wheel_radius', 0.076).value
        self.Lx = self.declare_parameter('half_wheelbase', 0.18).value
        self.Ly = self.declare_parameter('half_track', 0.20).value
        self.timeout = self.declare_parameter('cmd_timeout', 0.5).value
        self.max_wheel_speed = self.declare_parameter('max_wheel_speed', 12.0).value
        if not all(math.isfinite(v) and v > 0 for v in
                   (self.r, self.Lx, self.Ly, self.timeout, self.max_wheel_speed)):
            raise ValueError('Geometry, timeout and wheel speed limit must be positive and finite')
        # A paused /clock must not prevent stale commands from being cleared.
        self.watchdog_clock = Clock(clock_type=ClockType.STEADY_TIME)
        self.last_command = None
        self.sub = self.create_subscription(Twist, '/cmd_vel', self.cmd_cb, 10)
        self.pub_fl = self.create_publisher(Float64MultiArray, '/front_left_velocity_controller/commands', 10)
        self.pub_fr = self.create_publisher(Float64MultiArray, '/front_right_velocity_controller/commands', 10)
        self.pub_rl = self.create_publisher(Float64MultiArray, '/rear_left_velocity_controller/commands', 10)
        self.pub_rr = self.create_publisher(Float64MultiArray, '/rear_right_velocity_controller/commands', 10)

        self.watchdog = self.create_timer(
            0.05, self.watchdog_cb, clock=self.watchdog_clock)

    def publish_wheels(self, values):
        for publisher, value in zip(
                (self.pub_fl, self.pub_fr, self.pub_rl, self.pub_rr), values):
            publisher.publish(Float64MultiArray(data=[value]))

    def watchdog_cb(self):
        now = self.watchdog_clock.now().nanoseconds
        if self.last_command is None or (now - self.last_command) * 1e-9 >= self.timeout:
            self.publish_wheels([0.0] * 4)

    def cmd_cb(self, msg):
        vx = msg.linear.x
        vy = msg.linear.y
        omega = msg.angular.z
        if not all(math.isfinite(v) for v in (vx, vy, omega)):
            self.last_command = None
            self.publish_wheels([0.0] * 4)
            return
        self.last_command = self.watchdog_clock.now().nanoseconds
        w_fl = (vx - vy - (self.Lx + self.Ly) * omega) / self.r
        w_fr = (vx + vy + (self.Lx + self.Ly) * omega) / self.r
        w_rl = (vx + vy - (self.Lx + self.Ly) * omega) / self.r
        w_rr = (vx - vy + (self.Lx + self.Ly) * omega) / self.r
        values = [w_fl, w_fr, w_rl, w_rr]
        if not all(math.isfinite(v) for v in values):
            self.last_command = None
            self.publish_wheels([0.0] * 4)
            return
        # Scale all four wheels together to preserve the requested twist direction.
        scale = max(1.0, max(abs(v) for v in values) / self.max_wheel_speed)
        self.publish_wheels([v / scale for v in values])


def main():
    rclpy.init()
    node = MecanumKinematics()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        if rclpy.ok():
            node.publish_wheels([0.0] * 4)
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Wheel-feedback odometry for an X-layout mecanum base."""
import math

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import TransformStamped
from nav_msgs.msg import Odometry
from sensor_msgs.msg import JointState
from tf2_ros import TransformBroadcaster


class MecanumOdometry(Node):
    JOINTS = tuple(f'{corner}_wheel_joint' for corner in
                   ('front_left', 'front_right', 'rear_left', 'rear_right'))

    def __init__(self):
        super().__init__('mecanum_odometry')
        self.radius = self.declare_parameter('wheel_radius', 0.076).value
        self.lever = (self.declare_parameter('half_wheelbase', 0.18).value +
                      self.declare_parameter('half_track', 0.20).value)
        if self.radius <= 0 or self.lever <= 0:
            raise ValueError('Wheel radius and summed half distances must be positive')
        self.odom_frame = self.declare_parameter('odom_frame', 'odom').value
        self.base_frame = self.declare_parameter('base_frame', 'base_footprint').value
        self.x = self.y = self.yaw = 0.0
        self.previous_stamp = None
        self.publisher = self.create_publisher(Odometry, 'odom', 10)
        self.broadcaster = TransformBroadcaster(self)
        self.subscription = self.create_subscription(
            JointState, 'joint_states', self.joint_callback, 10)

    def joint_callback(self, msg):
        # Resolve by name: joint_state_broadcaster does not promise wheel order.
        try:
            wheels = [msg.velocity[msg.name.index(name)] for name in self.JOINTS]
        except (ValueError, IndexError):
            return
        if not all(math.isfinite(value) for value in wheels):
            return
        stamp = msg.header.stamp
        seconds = stamp.sec + stamp.nanosec * 1e-9
        fl, fr, rl, rr = wheels
        vx = self.radius * (fl + fr + rl + rr) / 4.0
        vy = self.radius * (-fl + fr + rl - rr) / 4.0
        wz = self.radius * (-fl + fr - rl + rr) / (4.0 * self.lever)
        if self.previous_stamp is not None:
            dt = seconds - self.previous_stamp
            if dt == 0:
                return
            if dt < 0:
                # Gazebo reset: start a new odometry epoch, never integrate backwards.
                self.x = self.y = self.yaw = 0.0
            else:
                self.x, self.y, self.yaw = integrate_pose(
                    self.x, self.y, self.yaw, vx, vy, wz, dt)
        self.previous_stamp = seconds

        odom = Odometry()
        odom.header.stamp = stamp
        odom.header.frame_id = self.odom_frame
        odom.child_frame_id = self.base_frame
        odom.pose.pose.position.x = self.x
        odom.pose.pose.position.y = self.y
        odom.pose.pose.orientation.z = math.sin(self.yaw / 2.0)
        odom.pose.pose.orientation.w = math.cos(self.yaw / 2.0)
        odom.twist.twist.linear.x = vx
        odom.twist.twist.linear.y = vy
        odom.twist.twist.angular.z = wz
        # Provisional wheel-odometry uncertainties; calibrate against measured slip.
        covariance = [0.02, 0.02, 1e6, 1e6, 1e6, 0.05]
        for i, variance in enumerate(covariance):
            odom.pose.covariance[i * 7] = variance
            odom.twist.covariance[i * 7] = variance
        self.publisher.publish(odom)
        transform = TransformStamped()
        transform.header = odom.header
        transform.child_frame_id = self.base_frame
        transform.transform.translation.x = self.x
        transform.transform.translation.y = self.y
        transform.transform.rotation = odom.pose.pose.orientation
        self.broadcaster.sendTransform(transform)


def integrate_pose(x, y, yaw, vx, vy, wz, dt):
    """Integrate constant body twist exactly, including simultaneous rotation."""
    angle = wz * dt
    if abs(angle) < 1e-8:
        a, b = dt, wz * dt * dt / 2.0
    else:
        a, b = math.sin(angle) / wz, (1.0 - math.cos(angle)) / wz
    dx, dy = a * vx - b * vy, b * vx + a * vy
    x += math.cos(yaw) * dx - math.sin(yaw) * dy
    y += math.sin(yaw) * dx + math.cos(yaw) * dy
    yaw = math.atan2(math.sin(yaw + angle), math.cos(yaw + angle))
    return x, y, yaw


def main(args=None):
    rclpy.init(args=args)
    node = MecanumOdometry()
    try:
        rclpy.spin(node)
```

## 13. 独立短程底盘诊断脚本

源文件：[scripts/validate_odometry.py](/home/deng/omove_ws/src/omove_description/scripts/validate_odometry.py)

```python
#!/usr/bin/env python3
"""Opt-in world01 wheel odometry diagnostic, using a separate simulation.

Sends short low-speed forward, lateral and rotation commands. Compares wheel
odometry with Gazebo ground truth; never uses ground truth to drive odometry.
The original world and existing maps are not modified.
"""
import argparse
import json
import math
import os
from pathlib import Path
import signal
import subprocess
import tempfile
import time
import xml.etree.ElementTree as ET


def wrap(angle):
    return math.atan2(math.sin(angle), math.cos(angle))


def relative(start, end):
    dx, dy = end[0] - start[0], end[1] - start[1]
    c, s = math.cos(start[2]), math.sin(start[2])
    return [c * dx + s * dy, -s * dx + c * dy, wrap(end[2] - start[2])]


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output', default='/tmp/omove_odometry_validation.json')
    parser.add_argument('--domain', type=int, default=96)
    parser.add_argument('--port', type=int, default=11359)
    parser.add_argument('--phase', choices=['all', 'forward', 'lateral', 'rotation'], default='all')
    parser.add_argument('--rotation-speed', type=float, default=.2)
    parser.add_argument('--controller-rate', type=int, default=None,
                        help='Test an alternate controller update rate using a temporary YAML')
    parser.add_argument('--velocity-pid', action='store_true',
                        help='Experiment with native Gazebo velocity PID using a temporary URDF')
    parser.add_argument('--wheel-control-mode', choices=['velocity', 'velocity_pid'],
                        default='velocity_pid', help='Control mode for the production robot model')
    args = parser.parse_args()
    if not math.isfinite(args.rotation_speed) or not 0 < args.rotation_speed <= .6:
        parser.error('--rotation-speed must be finite and in (0, 0.6] rad/s')
    if args.controller_rate is not None and not 1 <= args.controller_rate <= 1000:
        parser.error('--controller-rate must be in [1, 1000] Hz')
    output = Path(args.output).resolve()
    output.parent.mkdir(parents=True, exist_ok=True)
    os.environ.update(ROS_DOMAIN_ID=str(args.domain),
                      GAZEBO_MASTER_URI=f'http://localhost:{args.port}',
                      ROS_LOG_DIR=str(output.with_suffix('')) + '_roslogs')

    import rclpy
    from ament_index_python.packages import get_package_share_directory
    from geometry_msgs.msg import Twist
    from gazebo_msgs.msg import ModelStates
    from nav_msgs.msg import Odometry
    from sensor_msgs.msg import LaserScan, JointState
    from std_msgs.msg import Float64MultiArray
    from rclpy.qos import qos_profile_sensor_data

    rclpy.init()
    node = rclpy.create_node('wheel_odometry_validation')
    state = {}
    corners = ['front_left', 'front_right', 'rear_left', 'rear_right']
    samples = []
    publisher = node.create_publisher(Twist, '/cmd_vel', 10)

    def pose(p):
        q = p.orientation
        return [p.position.x, p.position.y,
                math.atan2(2 * (q.w*q.z + q.x*q.y), 1 - 2*(q.y*q.y + q.z*q.z))]

    def truth(msg):
        if 'omove' in msg.name:
            p = msg.pose[msg.name.index('omove')]
            state['truth'] = pose(p)
            state['height'] = p.position.z
            q = p.orientation
            state['tilt_rad'] = math.acos(max(-1., min(1., 1-2*(q.x*q.x+q.y*q.y))))
            state['truth_received'] = time.monotonic()

    def odom(msg):
        state['odom'] = pose(msg.pose.pose)
        state['sim_time'] = msg.header.stamp.sec + msg.header.stamp.nanosec * 1e-9
        state['odom_received'] = time.monotonic()

    def scan(msg):
        state['clearance'] = min((v for v in msg.ranges if math.isfinite(v)), default=math.inf)

    def joints(msg):
        try:
            state['wheel_feedback'] = [msg.velocity[msg.name.index(c + '_wheel_joint')] for c in corners]
        except (ValueError, IndexError):
            pass

    subscriptions = [
        node.create_subscription(ModelStates, '/diagnostic/model_states', truth, 1),
        node.create_subscription(Odometry, '/odom', odom, 1),
        node.create_subscription(LaserScan, '/scan', scan, qos_profile_sensor_data),
        node.create_subscription(JointState, '/joint_states', joints, 1),
    ]
    for c in corners:
        subscriptions.append(node.create_subscription(
            Float64MultiArray, '/' + c + '_velocity_controller/commands',
            lambda msg, c=c: state.setdefault('wheel_command', {}).update({c: list(msg.data)}), 1))

    def drive(vx, vy, wz, seconds):
        start = state['sim_time']
        deadline = time.monotonic() + 80
        msg = Twist()
        msg.linear.x, msg.linear.y, msg.angular.z = vx, vy, wz
        while state['sim_time'] - start < seconds:
            if time.monotonic() > deadline:
                raise TimeoutError('simulation paused or too slow')
            if any(time.monotonic() - state.get(k, 0) > 2
                   for k in ('truth_received', 'odom_received')):
                raise RuntimeError('truth or wheel odometry became stale')
            if (vx or vy or wz) and state.get('clearance', 0) < 0.45:
                raise RuntimeError(f'obstacle/self-return too close; stopping diagnostic: {state}')
            publisher.publish(msg)
            # Drain the high-rate odom/truth callbacks each control cycle.
            # One spin per sleep can let receive queues skew the comparison.
            for _ in range(20):
                rclpy.spin_once(node, timeout_sec=0.)
            if vx or vy or wz:
                samples.append({k: state.get(k) for k in
                                ['sim_time', 'truth', 'odom', 'wheel_feedback']})
                samples[-1]['wheel_command'] = dict(state.get('wheel_command', {}))
            time.sleep(0.02)

    report = {'phases': [], 'purpose': 'diagnostic, not navigation acceptance',
              'position_limit_m': 0.05, 'heading_limit_rad': 0.05,
              'note': 'Limits are diagnostic thresholds, not calibrated accuracy guarantees.',
              'success': False}
    with tempfile.TemporaryDirectory(prefix='omove_odom_') as temp:
        source = Path(get_package_share_directory('omove_description')) / 'worlds/world01/worlds/playground.world'
        world = Path(temp) / 'diagnostic.world'
        tree = ET.parse(source)
        plugin = ET.SubElement(tree.getroot().find('world'), 'plugin',
                               name='diagnostic_state', filename='libgazebo_ros_state.so')
        ET.SubElement(ET.SubElement(plugin, 'ros'), 'namespace').text = '/diagnostic'
        ET.SubElement(plugin, 'update_rate').text = '50'
        tree.write(world)
        launch_overrides = [f'wheel_control_mode:={args.wheel_control_mode}']
        report['wheel_control_mode'] = args.wheel_control_mode
        if args.controller_rate is not None:
            import yaml
            path = Path(get_package_share_directory('omove_description')) / 'config/mecanum_controllers.yaml'
            config = yaml.safe_load(path.read_text())
            config['controller_manager']['ros__parameters']['update_rate'] = args.controller_rate
            controllers = Path(temp) / 'controllers.yaml'
            controllers.write_text(yaml.safe_dump(config))
            launch_overrides.append(f'controllers_file:={controllers}')
        report['controller_rate_override'] = args.controller_rate
        report['velocity_pid_experiment'] = args.velocity_pid
        if args.velocity_pid:
            import xacro
            model = Path(get_package_share_directory('omove_description')) / 'urdf/omove.urdf.xacro'
            robot = ET.fromstring(xacro.process_file(str(model), mappings={'wheel_control_mode': 'velocity'}).toxml())
            for joint in robot.findall('ros2_control/joint'):
                interface = joint.find('command_interface')
                if interface is not None:
                    interface.set('name', 'velocity_pid')
                    for name, value in [('vel_kp', '.5'), ('vel_ki', '.5'), ('vel_kd', '0'),
                                        ('vel_max_integral_error', '2'), ('vel_min_integral_error', '-2')]:
                        ET.SubElement(joint, 'param', name=name).text = value
            robot_file = Path(temp) / 'velocity_pid.urdf'
            ET.ElementTree(robot).write(robot_file)
            launch_overrides.append(f'robot_file:={robot_file}')
        with output.with_suffix('.log').open('w') as log:
            process = subprocess.Popen([
                'ros2', 'launch', 'omove_description', 'world01_sim.launch.py',
                'slam:=false', 'gui:=false', f'world:={world}'] + launch_overrides,
                stdout=log, stderr=log, start_new_session=True)
            try:
                deadline = time.monotonic() + 70
                while not all(k in state for k in ('truth', 'odom', 'clearance')) or publisher.get_subscription_count() == 0:
                    if process.poll() is not None or time.monotonic() > deadline:
                        raise RuntimeError('simulation not ready; inspect log')
                    rclpy.spin_once(node, timeout_sec=0.05)
                drive(0., 0., 0., 1.0)
                for name, command, duration in [
                    ('forward', (.1, 0., 0.), 3.),
                    ('lateral', (0., .1, 0.), 3.),
                    ('rotation', (0., 0., args.rotation_speed), 3.),
                ]:
                    if args.phase not in ('all', name):
                        continue
                    samples.clear()
                    before_truth, before_odom = list(state['truth']), list(state['odom'])
                    drive(*command, duration)
                    drive(0., 0., 0., 1.5)
                    t = relative(before_truth, state['truth'])
                    o = relative(before_odom, state['odom'])
                    entry = dict(name=name, command=list(command), sim_seconds=duration,
                                 truth_delta=t, odom_delta=o,
                                 end_truth=list(state['truth']), height=state['height'],
                                 tilt_rad=state['tilt_rad'], scan_min=state['clearance'],
                                 position_error_m=math.hypot(t[0]-o[0], t[1]-o[1]),
                                 heading_error_rad=abs(wrap(t[2]-o[2])))
                    entry['samples'] = list(samples)
                    entry['within_diagnostic_limits'] = (entry['position_error_m'] <= .05
                                                        and entry['heading_error_rad'] <= .05)
                    report['phases'].append(entry)
                    print(json.dumps({k: v for k, v in entry.items() if k != 'samples'}), flush=True)
                report['success'] = all(p['within_diagnostic_limits'] for p in report['phases'])
            except Exception as error:
                report['error'] = f'{type(error).__name__}: {error}'
            finally:
                publisher.publish(Twist())
                output.write_text(json.dumps(report, indent=2))
                if process.poll() is None:
                    os.killpg(process.pid, signal.SIGINT)
                    try:
                        process.wait(timeout=15)
                    except subprocess.TimeoutExpired:
                        os.killpg(process.pid, signal.SIGTERM)
                        try:
                            process.wait(timeout=10)
                        except subprocess.TimeoutExpired:
                            os.killpg(process.pid, signal.SIGKILL)
                            process.wait()
                node.destroy_node()
                rclpy.shutdown()
    print('Report:', output, flush=True)
    if not report['success']:
        raise SystemExit(1)


if __name__ == '__main__':
    main()
```
